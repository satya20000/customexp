// Add each measure individually. Percentages are fractions; format 0.0%.
Total Orders = COUNTROWS(fact_order)
Delivered Orders = SUM(fact_order[delivered])
Cancelled Orders = SUM(fact_order[cancelled])
Cancellation Rate = DIVIDE([Cancelled Orders], [Total Orders])
Eligible Deliveries = SUM(fact_order[eligible_delivery])
Late Deliveries = SUM(fact_order[late])
Late Delivery Rate = DIVIDE([Late Deliveries], [Eligible Deliveries])
On Time Delivery Rate = IF([Eligible Deliveries] > 0, 1 - [Late Delivery Rate], BLANK())
Average Delivery Days = AVERAGE(fact_order[delivery_days])
Average Processing Days = AVERAGE(fact_order[processing_days])
Average Shipping Days = AVERAGE(fact_order[shipping_days])
Average Delay Days = CALCULATE(AVERAGE(fact_order[delay_days]), fact_order[late] = 1)
Average Review Score = AVERAGE(fact_order[review_score])
Reviewed Orders = COUNT(fact_order[review_score])
One Star Rate = DIVIDE(CALCULATE([Reviewed Orders], fact_order[review_score] = 1), [Reviewed Orders])
Five Star Rate = DIVIDE(CALCULATE([Reviewed Orders], fact_order[review_score] = 5), [Reviewed Orders])
Average Freight = AVERAGE(fact_order[freight])
Item AOV = AVERAGE(fact_order[gmv])
Bridge Orders = CALCULATE([Total Orders], TREATAS(VALUES(bridge_order_seller_category[order_id]), fact_order[order_id]))
Bridge Late Rate = CALCULATE([Late Delivery Rate], TREATAS(VALUES(bridge_order_seller_category[order_id]), fact_order[order_id]))
Bridge Review Score = CALCULATE([Average Review Score], TREATAS(VALUES(bridge_order_seller_category[order_id]), fact_order[order_id]))
Bridge Processing Days = CALCULATE([Average Processing Days], TREATAS(VALUES(bridge_order_seller_category[order_id]), fact_order[order_id]))
Bridge Shipping Days = CALCULATE([Average Shipping Days], TREATAS(VALUES(bridge_order_seller_category[order_id]), fact_order[order_id]))
